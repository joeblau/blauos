add_darwin_executable(blauos_guest_probe MACOSX_VERSION_MIN 11.0)
get_target_property(_probe_options blauos_guest_probe COMPILE_OPTIONS)
if(_probe_options)
  list(REMOVE_ITEM _probe_options -nostdinc)
  set_target_properties(blauos_guest_probe PROPERTIES COMPILE_OPTIONS "${_probe_options}")
endif()
target_sources(blauos_guest_probe PRIVATE guest_probe.c)
target_compile_options(blauos_guest_probe PRIVATE -isysroot ${PUREDARWIN_MACOSX_SDK} -fno-builtin -fno-stack-protector)
target_link_options(blauos_guest_probe PRIVATE -nostdlib -Wl,-L,$<TARGET_FILE_DIR:libSystem_B_stub> -Wl,-dylinker_install_name,/usr/lib/dyld -Wl,-platform_version,macos,11.0,11.5 -lSystem)
add_dependencies(blauos_guest_probe libSystem_B_stub dyld)
install(TARGETS blauos_guest_probe DESTINATION bin COMPONENT BaseSystem)
