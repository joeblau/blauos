function(add_blauos_probe name source)
  add_darwin_executable(${name} MACOSX_VERSION_MIN 11.0)
  get_target_property(_probe_options ${name} COMPILE_OPTIONS)
  if(_probe_options)
    list(REMOVE_ITEM _probe_options -nostdinc)
    set_target_properties(${name} PROPERTIES COMPILE_OPTIONS "${_probe_options}")
  endif()
  target_sources(${name} PRIVATE ${source})
  target_compile_options(${name} PRIVATE -isysroot ${PUREDARWIN_MACOSX_SDK} -fno-builtin -fno-stack-protector)
  target_link_options(${name} PRIVATE -nostdlib -Wl,-L,$<TARGET_FILE_DIR:libSystem_B_stub> -Wl,-dylinker_install_name,/usr/lib/dyld -Wl,-platform_version,macos,11.0,11.5 -lSystem)
  add_dependencies(${name} libSystem_B_stub dyld)
  install(TARGETS ${name} DESTINATION bin COMPONENT BaseSystem)
endfunction()

add_blauos_probe(blauos_guest_probe guest_probe.c)
add_blauos_probe(blauos_runtime_probe runtime_probe.c)
target_compile_options(blauos_runtime_probe PRIVATE -fblocks)
add_blauos_probe(blauos_bootstrap_probe bootstrap_probe.c)
add_blauos_probe(blauos_shutdown_probe shutdown_probe.c)
